\section{Review for Single Variable PDEs}
\label{sec:method}

Throughout, $\ddx u_j = (u_{j+1}-2u_j+u_{j-1})/\dx^2$ is the centered second difference and refinement means $\dx\to0$ with $\dt$ slaved to $\dx$.

\subsection{The scalar family and the optimal time step}

Consider the deterministic family
\begin{equation}\label{eq:family} u_t = D\,u_{xx} + \alpha\,u, \qquad D>0,\ \alpha\in\R,
\end{equation} whose stochastic motivation (the moment equations of the linear SPDE of the companion \secStochX, with $\alpha = \frac{k(k-1)}2\sigma^2$) is deferred there. The pure diffusion case $\alpha=0$ is exactly Chu's setting \citep{chu2011}.

Apply forward Euler in time and the three-point stencil in space. Taylor expansion of both sides, followed by repeated differentiation of
\eqref{eq:family} to convert time derivatives of $u$ into space derivatives (the operators $D\partial_{xx}$ and $\alpha$ commute, giving the binomial identity $u_{t^k} = (D\partial_{xx}+\alpha)^k u$), yields the local truncation error organized by derivative type:
\begin{equation}\label{eq:lte}
\mathcal{T} \;=\;
\underbrace{\Bigl(\tfrac{D^2\dt}2 - \tfrac{D\dx^2}{12}\Bigr)u_{xxxx}}_{\text{Group 1}}
\;+\;
\underbrace{\alpha D\,\dt\,u_{xx}}_{\text{Group 2}}
\;+\;
\underbrace{\tfrac{\alpha^2\dt}2\,u}_{\text{Group 3}}
\;+\;O(\dt^2,\dx^4).
\end{equation} Only Group 1 couples $\dt$ to $\dx$. Setting it to zero:

\begin{theorem}[Optimal time step {\citep{chu2011}}]\label{thm:ots}
For \eqref{eq:family} discretized by forward Euler / centered differences, the
unique step size annihilating the leading ($u_{xxxx}$) truncation term is
\begin{equation}\label{eq:dtopt2}
\boxed{\ \dto = \frac{\dx^2}{6D}\ }
\end{equation}
and no single step annihilates both the $u_{xxxx}$ and $u_{xxxxxx}$ truncation
terms, since the level-2 condition is $\dt = \dx^2/(\sqrt{60}\,D) \ne \dto$.
\end{theorem}
\noindent (Proof: one-line cancellation in \eqref{eq:lte}; uniqueness by the
level-2 computation. Full algebra in \texttt{sign-change.tex} \S1.)

\begin{proposition}[$\alpha$-independence]\label{prop:alpha}
$\dto$ is independent of the reaction parameter $\alpha$: the cancellation
operates on the diffusive group alone. For the stochastic hierarchy this makes
$\dto$ independent of the noise amplitude $\sigma$ and of the moment order $k$.
\end{proposition}

Following the derivation inspecting the truncation error, at $\dt=\dto$ the Group-1 term vanishes and the surviving residual is
\begin{equation}\label{eq:resid}
\mathcal{T}\big|_{\dto} \;=\;
\frac{\alpha\dx^2}6\,u_{xx} + \frac{\alpha^2\dx^2}{12D}\,u \;+\; O(\dx^4).
\end{equation}

\begin{remark}[Sign convention]\label{rem:sign}
The truncation error $\mathcal{T}$ as assembled in \eqref{eq:lte} is the \emph{forward} residual: $\text{exact} - \text{scheme} = \dt\,\mathcal{T} + \cdots$. Cancelling it therefore requires \emph{adding} $\dt\,\mathcal{T}$; all corrections below carry this sign.
\end{remark}

\begin{lemma}[Closed form correction term]The corrected factor at any given time step is,
\begin{equation}\label{eq:C1}
C^{(1)}_j = +\frac{\alpha\,\dx^4}{36D}\,\ddx u^n_j \;+\; \frac{\alpha^2\,\dx^4}{72D^2}\,u^n_j.
\end{equation}
\end{lemma}

\begin{theorem}[First-level accuracy]\label{thm:p1}

The corrected scheme
\begin{equation}\label{eq:scheme1} u^{n+1}_j = u^n_j + \dto\Bigl[D\,\ddx u^n_j + \alpha\,u^n_j\Bigr] + C^{(1)}_j
\end{equation} has step error $O(\dx^6)$ and global order $4$: the correction $C^{(1)}$ is itself $O(\dx^4)$, so its evaluation with the second-order operator $\ddx$ contributes only $O(\dx^6)$ to the step --- the three-point stencil is mathematically sufficient at $p=1$.
\end{theorem}

\subsection{Stability of Single Variable Systems}
\label{sec:stability}

Set $u^n_j = G^n e^{i\kappa j\dx}$; the three-point Laplacian has symbol
$\ddx u^n_j \to -\frac{4}{\dx^2}s\,u^n_j$ with
$s=\sin^2(\kappa\dx/2)\in[0,1]$. Measure the step against the optimum,
$\tau:=\dto=\dx^2/(6D)$:
\[
r=\frac{\dt}{\tau}=\frac{6D\dt}{\dx^2},\qquad
\eta=\frac{\alpha\dx^2}{D},\qquad
a=\alpha\dt=\frac{r\eta}{6},
\]
where $r$ is the step ratio used throughout
\S\ref{sec:regime1}--\ref{sec:regime2} (there the forced step is $h=r\tau$
with mismatch $\varepsilon=(r-1)\tau$), $\eta$ is the step-independent
reaction strength, and the legacy diffusion number is $\mu=r/6$. The exact
semi-discrete one-step factor is $E(s)=e^{-\frac{2r}{3}s+a}$: since the
reaction grows, $|G|\le1$ is the wrong requirement, and we use the von
Neumann condition with growth allowance
\begin{equation}\label{eq:vN}
|G(s)|\le 1 + C\,\dt \quad\text{uniformly in } s\in[0,1],
\end{equation}
along refinement paths $\dt=r\tau$ with $r$ fixed (Lax--Richtmyer); the
sharper natural comparison is $|G(s)|\le e^{a}$.

We proceed in four steps: the amplification factors themselves
(Lemma~\ref{lem:factors}); stability at the optimal step, which is what
makes $\dto$ safe for every moment of the hierarchy
(Lemma~\ref{lem:optstab}); the grid-scale ceilings away from the optimum,
which fix the operating ranges of \S\ref{sec:regime1}--\ref{sec:regime2}
(Lemma~\ref{lem:cfl}); and the scope of the frozen correction
(Remark~\ref{rem:scope}). Theorem~\ref{prop:stability} collects the result.

\begin{lemma}[Amplification factors]\label{lem:factors}
With $r=\dt/\tau$ and $\eta=\alpha\dx^2/D$,
\begin{align}
G_{\mathrm{FE}} &= 1-\tfrac{2r}{3}s+\tfrac{r\eta}{6},\label{eq:gfe}\\
G_{\mathrm{NIDC}} &= 1-\tfrac{2r}{3}s+\tfrac{r\eta}{6}
-\tfrac{\eta}{9}s+\tfrac{\eta^2}{72},\label{eq:gnidc}
\end{align}
and $G_{\mathrm{NIDC}}=G_{\mathrm{FE}}-\frac{\eta}{9}s+\frac{\eta^2}{72}$
is strictly decreasing in $s$ on $[0,1]$.
\end{lemma}

\begin{proof}
Substituting the symbol of $\ddx$ into the uncorrected update gives
\eqref{eq:gfe}. For the corrected scheme, the $C^{(1)}$ terms of
\eqref{eq:C1} contribute
$\frac{\alpha\dx^4}{36D}\cdot\bigl(-\frac{4s}{\dx^2}\bigr)=-\frac{\eta}{9}s$
and $\frac{\alpha^2\dx^4}{72D^2}=\frac{\eta^2}{72}$, giving \eqref{eq:gnidc}
and the perturbation identity. The slope in $s$ is
$-\bigl(\frac{2r}{3}+\frac{\eta}{9}\bigr)<0$.
\end{proof}

\noindent Both perturbation terms are $O(\dx^2)$, so the extremes over
$s\in[0,1]$ sit at the endpoints $s=0$ (physical mode) and $s=1$ (grid
mode); every bound below is therefore an endpoint check.

\begin{lemma}[Stability at the optimal step]\label{lem:optstab}
At $r=1$ (so $a=\eta/6$),
\begin{equation}\label{eq:factor}
G_{\mathrm{NIDC}}(s)=\Bigl(1-\tfrac{2}{3}s\Bigr)(1+a)+\tfrac{a^2}{2},
\end{equation}
and for every $\eta\ge0$ and all $s\in[0,1]$,
\[
\tfrac13(1+a)+\tfrac{a^2}{2}\;\le\;G_{\mathrm{NIDC}}(s)\;\le\;
1+a+\tfrac{a^2}{2}\;\le\; e^{a}\qquad(a\ge0).
\]
In particular $G_{\mathrm{NIDC}}>0$ (no oscillatory modes) and
\eqref{eq:vN} holds with no restriction on $\eta$.
\end{lemma}

\begin{proof}
Substitute $r=1$ in \eqref{eq:gnidc}; $\eta=6a$ gives \eqref{eq:factor}.
Since $1-\frac23 s\in[\frac13,1]$, the bracket bounds follow, and
$1+a+\frac{a^2}{2}\le e^a$ is the tail of the exponential series. The lower
end is positive, giving the two-sided bound.
\end{proof}

\noindent This is the headline safety fact: the correction never
destabilizes the optimal step, at any reaction strength. The offset
$+\eta^2/72$ is not destabilizing --- it is what makes $G(0)$ match $e^a$
to $O(a^2)$. Plain forward Euler at the same $r$ has
$G_{\mathrm{FE}}(s)=(1-\frac23 s)+a$, inside $[\frac13+a,\,1+a]$.

\begin{lemma}[Grid-scale ceilings for general $r$]\label{lem:cfl}
For $0<\eta<8$, requiring the lower side of \eqref{eq:vN} at $s=1$, i.e.\
$G(1)\ge-(1+a)$, gives
\begin{align}
r_{\mathrm{FE},\max}&=\frac{6}{2-\eta}=\frac{3}{1-\eta/2},\label{eq:cflfe}\\
r_{\mathrm{NIDC},\max}&=\frac{3(2-\eta/9+\eta^2/72)}{2-\eta},\label{eq:cflnidc}
\end{align}
with ratio
\begin{equation}\label{eq:shrink2}
\frac{r_{\mathrm{NIDC},\max}}{r_{\mathrm{FE},\max}}
=1-\frac{\eta}{18}+\frac{\eta^2}{144}.
\end{equation}
\end{lemma}

\begin{proof}
At $s=1$ the factors are affine in $r$:
$G_{\mathrm{FE}}(1)=1-\frac{2r}{3}+\frac{r\eta}{6}$ and
$G_{\mathrm{NIDC}}(1)=G_{\mathrm{FE}}(1)-\frac{\eta}{9}+\frac{\eta^2}{72}$.
Setting each equal to $-(1+a)$ with $a=\frac{r\eta}{6}$ and solving the
linear equation in $r$ gives \eqref{eq:cflfe}--\eqref{eq:cflnidc}; the ratio
\eqref{eq:shrink2} is their quotient.
\end{proof}

\noindent Two directions matter and should not be conflated. The
\emph{shrink} is relative to forward Euler at the same $\eta$: $-1.6\%$ at
the coarse $\eta=0.3$ test point ($r_{\mathrm{FE},\max}=3.529$,
$r_{\mathrm{NIDC},\max}=3.473$, ratio $0.98396$), and $O(\dx^2)$. The \emph{absolute}
ceiling $\dt_{\max}=r_{\mathrm{NIDC},\max}\tau
=\frac{2-\eta/9+\eta^2/72}{2-\eta}\cdot\frac{\dx^2}{2D}$ loosens with
$\eta$ relative to the $\eta=0$ ceiling $\dx^2/(2D)$ ($+16\%$ at
$\eta=0.3$). (The stricter test $G(1)\ge-1$ gives ceilings $6/(4-\eta)$ and
$6(2-\eta/9+\eta^2/72)/(4-\eta)$, with the same ratio.) These ceilings are
exactly what \S\ref{sec:regime1} uses as its operating range ($r<4$) and
\S\ref{sec:regime2} as its switching rule ($r_c$).

\begin{remark}[Scope of the frozen correction]\label{rem:scope}
$C^{(1)}$ is derived at $\dt=\dto$ and is independent of $\dt$. At $s=0$,
\[ G_{\mathrm{NIDC}}(0)-e^{a}\;\approx\;\frac{\eta^2}{72}\bigl(1-r^2\bigr),
\]
which is $\le0$ iff $r\ge 1$. For $r<1$ the excess is $\dt$-independent, so
at fixed $\dx$ with $\dt\to0$ the accumulated growth
$(1+\text{const})^{T/\dt}$ diverges. Stability statements therefore refer
to refinement paths with $r$ fixed, where the excess is $O(\dt^2)$ and
\eqref{eq:vN} holds; the $\dt/\dto\in\{0.5,\dots,2\}$ table is a sweep over
such paths, not a fixed-$\dx$ study. The method is designed for $r=1$.
\end{remark}

\begin{corollary}[Pair reduction]\label{cor:pair}
For a pair of species with $D\to D_i+D_j$ and
$\alpha\to\alpha_{ij}=R_{ij}$, \eqref{eq:factor} holds verbatim with
$r_{ij}=1$ at $\dto^{(ij)}$.
\end{corollary}

\begin{theorem}[Von Neumann stability]\label{prop:stability}
With $s=\sin^2(\kappa\dx/2)$, $r=\dt/\tau$, $\eta=\alpha\dx^2/D$, the
factors are \eqref{eq:gfe}--\eqref{eq:gnidc}. At the optimal step $r=1$ the
corrected scheme is unconditionally stable for every $\eta\ge0$
(Lemma~\ref{lem:optstab}); for general $r$ and $0<\eta<8$ the grid-scale
bound tightens from \eqref{eq:cflfe} to \eqref{eq:cflnidc}, a relative
shrink of $1-\eta/18+O(\eta^2)$, i.e.\ $O(\dx^2)$, about $-1.6\%$ at
$\eta=0.3$. The optimal step $r=1$ sits a factor of at least $3$ inside
the CFL ceiling for both schemes.
\end{theorem}

\begin{proof}
Assemble Lemmas~\ref{lem:factors}, \ref{lem:optstab} and \ref{lem:cfl};
the endpoint reduction follows from the monotonicity in
Lemma~\ref{lem:factors}.
\end{proof}

\subsection{Efficiency profile}

The method's cost argument is architectural, not just arithmetic: versus multi-stage Runge--Kutta, OTS-NIDC is single-pass (one read, one write of the state vector; no stage storage, no stage barriers); versus compact schemes, it is strictly local (no tridiagonal solves), so it maps onto GPUs and bandwidth-bound hardware without global communication. At $N_x=256$ the verified scalar runs show $9.2\times10^{4}\times$ lower error than forward Euler at identical grid and step ($1.4\times10^{3}\times$ already at $N_x=32$) \tagV --- order wins over constants at refinement.
